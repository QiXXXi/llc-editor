// 创建独立模块的函数
function createTextSkillModule(container) {
    var imgCache = null;
    var isLoading = false;
    var callbacks = [];

    // 从data属性获取配置
    var colorName = container.getAttribute('data-color') || '暴怒';
    var text = container.getAttribute('data-text') || '技能';
    var selectedColor = COLOR_MAP[colorName] || '#A0392B';
    
    // 设置唯一ID
    var containerId = 'textskill-container-' + new Date().getTime() + '-' + Math.random().toString(36).substr(2, 9);
    container.id = containerId;

    // 加载并缓存图片
    function loadImg(cb) {
        if (imgCache) { cb(imgCache); return; }
        if (isLoading) { callbacks.push(cb); return; }
        
        isLoading = true;
        var img = new Image();
        img.crossOrigin = "anonymous";
        img.src = 'https://huiji-public.huijistatic.com/limbuscompany/uploads/d/da/BattleUI_CoinPopup_skillName.png';
        
        callbacks.push(cb);
        
        img.onload = function() {
            imgCache = img;
            isLoading = false;
            for (var i = 0; i < callbacks.length; i++) {
                callbacks[i](img);
            }
            callbacks = [];
        };
        
        img.onerror = function() {
            isLoading = false;
            for (var i = 0; i < callbacks.length; i++) {
                callbacks[i](null);
            }
            callbacks = [];
        };
    }

    // 应用颜色到图片
    function applyColor(img, color, callback) {
        var canvas = document.createElement('canvas');
        var ctx = canvas.getContext('2d');
        
        canvas.width = img.width;
        canvas.height = img.height;
        ctx.drawImage(img, 0, 0);
        
        var imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
        var data = imgData.data;
        
        // 解析目标颜色
        var hex = color.replace('#', '');
        var r = parseInt(hex.substring(0, 2), 16);
        var g = parseInt(hex.substring(2, 4), 16);
        var b = parseInt(hex.substring(4, 6), 16);
        
        // 处理每个像素
        for (var i = 0; i < data.length; i += 4) {
            var brightness = (data[i] + data[i+1] + data[i+2]) / 765;
            data[i] = r * brightness;
            data[i+1] = g * brightness;
            data[i+2] = b * brightness;
        }
        
        ctx.putImageData(imgData, 0, 0);
        
        var coloredImg = new Image();
        coloredImg.onload = function() {
            callback(coloredImg);
        };
        coloredImg.src = canvas.toDataURL('image/png');
    }

    // 创建中间重复图案
    function createPattern(img, leftWidth, middleWidth, height) {
        var canvas = document.createElement('canvas');
        var ctx = canvas.getContext('2d');
        
        canvas.width = middleWidth;
        canvas.height = height;
        
        var scale = height / img.height;
        
        ctx.drawImage(
            img,
            leftWidth / scale, 0, middleWidth / scale, img.height,
            0, 0, middleWidth, height
        );
        
        return canvas.toDataURL('image/png');
    }

    // 更新DOM元素
    function updateElements(scaledW, leftW, middleW, rightW, height, img) {
        var left = container.querySelector('.textskill-left');
        var middle = container.querySelector('.textskill-middle-repeat');
        var right = container.querySelector('.textskill-right');
        
        if (!left || !middle || !right) {
            console.warn('容器结构不完整，跳过更新');
            return;
        }
        
        left.style.width = leftW + 'px';
        left.style.backgroundSize = scaledW + 'px 100%';
        
        right.style.width = rightW + 'px';
        right.style.backgroundSize = scaledW + 'px 100%';
        right.style.backgroundPosition = '-' + (leftW + middleW) + 'px 0';
        
        if (img && img.src) {
            left.style.backgroundImage = 'url(' + img.src + ')';
            right.style.backgroundImage = 'url(' + img.src + ')';
        }
        
        if (img) {
            var pattern = createPattern(img, leftW, middleW, height);
            middle.style.backgroundImage = 'url(' + pattern + ')';
            middle.style.backgroundSize = middleW + 'px 100%';
        }
    }

    // 调整容器宽度
    function adjustWidth(rightW) {
        var textElem = container.querySelector('.textskill-text');
        var middleContainer = container.querySelector('.textskill-middle-container');
        var middle = container.querySelector('.textskill-middle-repeat');
        
        if (!textElem || !middleContainer || !middle) return;
        
        var textW = textElem.scrollWidth + 20;
        var middleW = Math.max(0, textW - rightW);
        
        middle.style.width = middleW + 'px';
        middleContainer.style.width = textW + 'px';
    }

    // 主初始化函数
    function init() {
        loadImg(function(img) {
            if (!img) {
                // 图片加载失败，使用默认尺寸
                var w = 96;
                var leftW = Math.round(w * 0.05);
                var middleW = Math.round(w * 0.65);
                var rightW = Math.round(w * 0.30);
                updateElements(w, leftW, middleW, rightW, 32, null);
                adjustWidth(rightW);
                return;
            }
            
            // 应用颜色
            applyColor(img, selectedColor, function(coloredImg) {
                // 计算尺寸
                var containerH = 32;
                var scale = containerH / coloredImg.height;
                var scaledW = coloredImg.width * scale;
                
                var leftW = Math.round(scaledW * 0.05);
                var middleW = Math.round(scaledW * 0.65);
                var rightW = Math.round(scaledW * 0.30);
                
                // 更新元素
                updateElements(scaledW, leftW, middleW, rightW, containerH, coloredImg);
                adjustWidth(rightW);
            });
        });
    }

    return {
        init: init,
        containerId: containerId,
        color: selectedColor,
        text: text
    };
}


mw.hook('wikipage.content').add(function($content) {

    var containers = $content[0].querySelectorAll('.textskill-container');
    var modules = [];
    
    for (var i = 0; i < containers.length; i++) {
        var container = containers[i];

        if (container.className.indexOf('initialized') === -1) {
            container.className += ' initialized';
            
            var module = createTextSkillModule(container);
            modules.push(module);
            module.init();
            
            console.log('初始化模块: ID=' + module.containerId + ', 颜色=' + module.color + ', 文本="' + module.text + '"');
        }
    }
    
    if (modules.length > 0) {
        console.log('共初始化 ' + modules.length + ' 个技能模块');
    }
});

( function () {
    'use strict';

    /* 防止皮肤/缓存导致重复执行 */
    if ( window.__personalitySkillInit ) { return; }
    window.__personalitySkillInit = true;

    var CLASSES = [
        'personalityskill1',
        'personalityskill2',
        'personalityskill3',
        'personalityskill4'
    ];
    var LABELS = [ '1技能', '2技能', '3技能', '守备技能' ];

    var COLOR_TEXT   = '#D6BA9A';
    var COLOR_BG     = '#090604';
    var COLOR_BORDER = '#D6BA9A';
    var COLOR_ACTIVE = '#FE9232';
    var COLOR_SHADOW = '#DB2C00';

var CSS =
    '.personalityskill-wrapper{display:block;margin:.5em 0 1em;line-height:1.7;}' +
    '.personalityskill-tabs{display:flex;flex-wrap:wrap;gap:4px;margin:0;padding:0;}' +
    '.personalityskill-tab{display:inline-block;padding:3px 14px;cursor:pointer;' +
        'color:' + COLOR_TEXT + ';background-color:' + COLOR_BG + ';' +
        'border:1px solid ' + COLOR_BORDER + ';white-space:nowrap;' +
        '-webkit-user-select:none;user-select:none;' +
        'transition:color .2s ease,text-shadow .2s ease;}' +
    '.personalityskill-tab:hover{color:' + COLOR_ACTIVE + ';' +
        'text-shadow:0 0 6px ' + COLOR_SHADOW + ';}' +
    '.personalityskill-tab.active{color:' + COLOR_ACTIVE + ';' +
        'text-shadow:0 0 6px ' + COLOR_SHADOW + ';}' +
    '.personalityskill-panels{margin-top:6px;}' +
    '.personalityskill-panel{display:none;color:' + COLOR_TEXT + ';}' +
    '.personalityskill-panel.active{display:block;}';
    
    var styleDone = false;
    function ensureCSS() {
        if ( styleDone ) { return; }
        styleDone = true;
        if ( window.mw && mw.util && typeof mw.util.addCSS === 'function' ) {
            mw.util.addCSS( CSS );   // 推荐方式
            return;
        }
        var s = document.createElement( 'style' );
        s.type = 'text/css';
        s.appendChild( document.createTextNode( CSS ) );
        ( document.head || document.getElementsByTagName( 'head' )[ 0 ] ).appendChild( s );
    }

    function indexOfClass( el ) {
        for ( var i = 0; i < CLASSES.length; i++ ) {
            if ( el.classList.contains( CLASSES[ i ] ) ) { return i; }
        }
        return -1;
    }

    function build( items ) {
        var parent = items[ 0 ].parentNode;
        if ( !parent ) { return; }

        var wrapper  = document.createElement( 'div' );
        wrapper.className = 'personalityskill-wrapper';

        var tabBar   = document.createElement( 'div' );
        tabBar.className = 'personalityskill-tabs';

        var panelBox = document.createElement( 'div' );
        panelBox.className = 'personalityskill-panels';

        var tabs   = [];
        var panels = [];

        items.forEach( function ( el, i ) {
            var idx = indexOfClass( el );

            var tab = document.createElement( 'span' );
            tab.className = 'personalityskill-tab';
            tab.textContent = LABELS[ idx ] || ( ( idx + 1 ) + '技能' );
            tab.setAttribute( 'role', 'button' );
            tab.setAttribute( 'tabindex', '0' );
            tabBar.appendChild( tab );

            var panel = document.createElement( 'div' );
            panel.className = 'personalityskill-panel';
            el.parentNode.insertBefore( panel, el );
            panel.appendChild( el );

            tabs.push( tab );
            panels.push( panel );

            function activate() {
                tabs.forEach( function ( t, j ) {
                    t.classList.toggle( 'active', j === i );
                } );
                panels.forEach( function ( p, j ) {
                    p.classList.toggle( 'active', j === i );
                } );
            }

            tab.addEventListener( 'click', activate );
            tab.addEventListener( 'keydown', function ( e ) {
                if ( e.key === 'Enter' || e.key === ' ' ) {
                    e.preventDefault();
                    activate();
                }
            } );
        } );

        parent.insertBefore( wrapper, panels[ 0 ] );
        wrapper.appendChild( tabBar );
        wrapper.appendChild( panelBox );

        panels.forEach( function ( p ) { panelBox.appendChild( p ); } );

        if ( tabs[ 0 ] ) {
            tabs[ 0 ].classList.add( 'active' );
            panels[ 0 ].classList.add( 'active' );
        }
    }

    function init() {
        ensureCSS();

        var selector = CLASSES.map( function ( c ) { return '.' + c; } ).join( ',' );
        var nodes = Array.prototype.slice.call( document.querySelectorAll( selector ) );
        if ( !nodes.length ) { return; }

        var groups = [];
        nodes.forEach( function ( el ) {
            var parent = el.parentNode;
            var g = null;
            for ( var i = 0; i < groups.length; i++ ) {
                if ( groups[ i ].parent === parent ) { g = groups[ i ]; break; }
            }
            if ( !g ) {
                g = { parent: parent, items: [] };
                groups.push( g );
            }
            g.items.push( el );
        } );

        groups.forEach( function ( g ) {
            g.items.sort( function ( a, b ) {
                return indexOfClass( a ) - indexOfClass( b );
            } );
            build( g.items );
        } );
    }

    if ( document.readyState === 'loading' ) {
        document.addEventListener( 'DOMContentLoaded', init );
    } else {
        init();
    }
}() );
